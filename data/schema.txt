CREATE TABLE User (ID INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL UNIQUE, Username TEXT NOT NULL UNIQUE,Password TEXT NOT NULL);
CREATE TABLE ListTask (ID INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL UNIQUE,[ID User] REFERENCES user (ID) NOT NULL,Name TEXT NOT NULL);
CREATE TABLE Task (ID INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL UNIQUE,[ID ListTask] REFERENCES ListTask (ID) NOT NULL,Name TEXT NOT NULL,State NUMERIC NOT NULL);
